    <h1>لعبة الذاكرة البرمجية</h1>
    <div class="score-board">المحاولات: <span id="moves-count">0</span></div>

    <div class="game-grid" id="grid"></div>

    <script>
        // الرموز أو الحروف المخفية داخل الكروت (كل رمز مكرر مرتين)
        const icons = ['A', 'A', 'B', 'B', 'C', 'C', 'D', 'D', 'E', 'E', 'F', 'F', 'G', 'G', 'H', 'H'];
        
        // خلط الكروت عشوائياً (Shuffle)
        icons.sort(() => 0.5 - Math.random());

        const grid = document.getElementById('grid');
        const movesDisplay = document.getElementById('moves-count');
        
        let firstCard = null;
        let secondCard = null;
        let lockBoard = false;
        let moves = 0;
        let matches = 0;

        // إنشاء الكروت داخل الشبكة
        function createBoard() {
            icons.forEach((icon, index) => {
                const card = document.createElement('div');
                card.classList.add('card');
                card.dataset.icon = icon;
                card.innerText = icon;
                card.addEventListener('click', flipCard);
                grid.appendChild(card);
            });
        }

        // دالة الضغط على الكرت
        function flipCard() {
            if (lockBoard) return;
            if (this === firstCard) return;
            if (this.classList.contains('matched')) return;

            this.classList.add('flipped');

            if (!firstCard) {
                firstCard = this;
                return;
            }

            secondCard = this;
            moves++;
            movesDisplay.innerText = moves;

            checkMatch();
        }

        // التحقق من تطابق الكرتين
        function checkMatch() {
            let isMatch = firstCard.dataset.icon === secondCard.dataset.icon;

            if (isMatch) {
                disableCards();
            } else {
                unflipCards();
            }
        }

        // في حال التطابق
        function disableCards() {
            firstCard.classList.add('matched');
            secondCard.classList.add('matched');
            matches++;

            resetTurn();

            // إذا تم إيجاد جميع الأزواج الـ 8
            if (matches === 8) {
                setTimeout(() => {
                    alert(`مبروك! لقد فزت باللعبة بعد ${moves} محاولة!`);
                }, 500);
            }
        }

        // في حال عدم التطابق (إعادة إغلاق الكروت بعد ثانية)
        function unflipCards() {
            lockBoard = true;

            setTimeout(() => {
                firstCard.classList.remove('flipped');
                secondCard.classList.remove('flipped');
                resetTurn();
            }, 1000);
        }

        // إعادة تهيئة المتغيرات للضغط القادم
        function resetTurn() {
            [firstCard, secondCard] = [null, null];
            lockBoard = false;
        }

        // تشغيل اللعبة عند فتح الصفحة
        createBoard();
    </script>
